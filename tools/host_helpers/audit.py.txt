"""Private command receipts. Call only for approved non-secret resource inputs.

Never use for credential commands, Secret resources, kubeconfig reads or env dumps.
The caller keeps its authorization/deadline/target checks; this adds no retries.
"""
import datetime as dt
import hashlib
import json
import os
from pathlib import Path
import subprocess
import uuid


def now():
    return dt.datetime.now(dt.timezone.utc).isoformat()


def digest(data):
    return hashlib.sha256(data).hexdigest()


def private_write(path, data):
    path = Path(path)
    tmp = path.with_name(path.name + '.' + uuid.uuid4().hex + '.tmp')
    with os.fdopen(os.open(tmp, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600), 'wb') as f:
        f.write(data)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp, path)
    fd = os.open(path.parent, os.O_RDONLY | os.O_DIRECTORY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def audited_run(directory, argv, *, target, input_bytes=None, timeout=8, runner=subprocess.run):
    """Persist intent before dispatch and outputs before returning/raising.

    Failure to persist the intent prevents execution. Timeout is unconfirmed,
    never retried. The subprocess timeout handler kills/reaps its direct child.
    No exception string is recorded (it may contain sensitive command context).
    """
    directory = Path(directory)
    directory.mkdir(mode=0o700, parents=True, exist_ok=True)
    if directory.is_symlink() or directory.stat().st_mode & 0o077:
        raise PermissionError('audit directory must be private')
    receipt = directory / uuid.uuid4().hex
    receipt.mkdir(mode=0o700)
    record = {'argv': list(argv), 'target': target, 'startedAt': now(),
              'stdinSHA256': digest(input_bytes) if input_bytes is not None else None,
              'stdinBytes': len(input_bytes) if input_bytes is not None else 0,
              'state': 'DISPATCHING', 'exitCode': None}
    def save():
        private_write(receipt / 'receipt.json', (json.dumps(record, indent=2) + '\n').encode())
    if input_bytes is not None:
        private_write(receipt / 'stdin.bin', input_bytes)
    save()
    stdout = stderr = b''
    try:
        result = runner(argv, input=input_bytes, capture_output=True, timeout=timeout)
        stdout, stderr = result.stdout, result.stderr
        record.update(exitCode=result.returncode,
                      state='RESPONSE_RECEIVED' if result.returncode == 0 else 'REJECTED_OR_UNCONFIRMED')
        return result, receipt
    except subprocess.TimeoutExpired as exc:
        stdout, stderr = exc.stdout or b'', exc.stderr or b''
        record.update(state='TIMEOUT_UNCONFIRMED', exceptionType=type(exc).__name__)
        raise
    except BaseException as exc:
        record.update(state='DISPATCH_EXCEPTION_UNCONFIRMED', exceptionType=type(exc).__name__)
        raise
    finally:
        # Independent receipt survives errors in the caller's subsequent parsing/save.
        private_write(receipt / 'stdout.bin', stdout)
        private_write(receipt / 'stderr.bin', stderr)
        record.update(completedAt=now(), stdoutSHA256=digest(stdout), stderrSHA256=digest(stderr))
        save()
